        let currentImage = productImage[0]

        arrows.addEventListener('click', (event) => {
        if (arrowRight.contains(event.target)) {
                const currentIndex = productImage.findIndex(cIndex => cIndex === currentImage )  

                const next = currentIndex + 1;

        if (next === (productImage.length - 1)) {
            arrowRight.style.display = 'none' 
                }

            currentImage = productImage[next]
                
            arrows.style.backgroundImage = `url('${currentImage}')`;
                
        }
        if (arrowLeft.contains(event.target)) {
            const  currentIndex = productImage.findIndex(cIndex => cIndex === currentImage)

            const next = currentIndex - 1;

            

            if (next === (productImage - 1)) {
                    arrowLeft.style.display = 'none'
                }

                currentImage = productImage[next]

                arrows.style.backgroundImage = `url('${currentImage}')`;

                arrowRight.style.display ='block'
            }
        })
    }
    carousel()

document.addEventListener('DOMContentLoaded', () => {
   const container = document.querySelector('.food-list');

 const API = async () => {
    const url = './data.json';

    try {
        
    const fetchAPI = await fetch(url);
    const response = await fetchAPI.json()
    console.log(response);

    getFoodList(response)
    } catch (error) {
        console.log('this is an error '+ error);
        
    }
 }
 API()

 const getFoodList = (response) => {

    let HTMLContent = '';

    const foodItem = response.map((element) => {
        const imgPath = element.image.desktop.replace('/assets','');
        
        const foodlistHTMLStructure = `
        <article class="food-item">
            <div class="food-container">
                <figure class="food-img">
                    <img src=${imgPath} alt="${element.name}">
                </figure>
                <figcaption>
                    <img src="./images/icon-add-to-cart.svg" alt="">
                    <span>Add to Cart</span>
                </figcaption>
            </div>
            <div class="food-details">
                <p>${element.category}</p>
                <h3 class="title">${element.name}</h3>
                <p class="price">$${element.price.toFixed(2)}</p>
            </div>
        </article>
    `;

    HTMLContent += foodlistHTMLStructure
    })
    container.innerHTML = HTMLContent;
 }

    
 
});
